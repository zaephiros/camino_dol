/** * ===== 양방향 동기화용 웹앱 (app_sync 필드 쓰기 + 다이어리 읽기/쓰기) ===== * * 이 웹앱 하나로 두 가지 역할을 합니다: * 1) 일자별 화면에서 입력한 예약/교통 메모를 app_sync 시트에 씀 (기존 기능) * 2) 메인 화면의 "오늘의 다이어리"를 diary 시트(날짜별 한 행)에 읽고 씀 (신규) * * diary 시트는 처음 쓰기가 들어올 때 자동으로 생성됩니다 (헤더: date, memo). * 별도로 시트를 만들거나 "웹에 게시"를 다시 할 필요 없습니다 — 이 웹앱 URL 하나로 다 됩니다. * * ===== 배포 방법 (처음 설치거나, 코드를 바꿨을 때) ===== * 1) Apps Script 편집기 상단 [배포 > 배포 관리] * 2) 이미 배포한 게 있으면 연필(수정) 아이콘 -> 버전: "새 버전" -> 배포 * (처음이라면 [배포 > 새 배포] -> 유형: 웹 앱 -> 실행: 나 -> 액세스: 전체) * 3) 웹 앱 URL을 앱의 "동기화 설정" 화면 쓰기 URL 칸에 붙여넣기 (URL은 재배포해도 안 바뀝니다) */ const DIARY_SHEET_NAME = 'diary'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); // { sheet, date, field, value } const sheetName = String(data.sheet || 'app_sync').trim(); const date = String(data.date || '').trim(); const field = String(data.field || (sheetName === DIARY_SHEET_NAME ? 'memo' : '')).trim(); const value = data.value === undefined || data.value === null ? '' : String(data.value); if (!date || !field) return jsonResponse({ ok: false, error: 'date/field가 필요합니다' }); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(sheetName); if (!sheet && sheetName === DIARY_SHEET_NAME) { // 다이어리 시트가 없으면 자동 생성 sheet = ss.insertSheet(DIARY_SHEET_NAME); sheet.getRange(1, 1, 1, 2).setValues([['date', 'memo']]); } if (!sheet) return jsonResponse({ ok: false, error: sheetName + ' 시트를 찾을 수 없습니다' }); const row = upsertCell(sheet, date, field, value); return jsonResponse({ ok: true, sheet: sheetName, date, field, row }); } catch (err) { return jsonResponse({ ok: false, error: String(err) }); } } function doGet(e) { try { const sheetName = String((e.parameter && e.parameter.sheet) || DIARY_SHEET_NAME).trim(); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return jsonResponse({ ok: true, rows: [] }); // 아직 시트가 없으면 빈 목록 const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow < 2) return jsonResponse({ ok: true, rows: [] }); const header = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h).trim()); const values = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); const rows = values.map(r => { const obj = {}; header.forEach((h, idx) => { obj[h] = r[idx]; }); return obj; }).filter(r => r.date); return jsonResponse({ ok: true, rows }); } catch (err) { return jsonResponse({ ok: false, error: String(err) }); } } /** date를 키로 하는 행을 찾아 field 컬럼에 value를 쓴다. 없으면 행/컬럼을 새로 만든다. */ function upsertCell(sheet, date, field, value) { const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); const headerRow = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h).trim()); let dateColIdx = headerRow.indexOf('date'); if (dateColIdx === -1) { sheet.getRange(1, 1).setValue('date'); dateColIdx = 0; } let fieldColIdx = headerRow.indexOf(field); if (fieldColIdx === -1) { fieldColIdx = Math.max(lastCol, dateColIdx + 1); sheet.getRange(1, fieldColIdx + 1).setValue(field); } let targetRow = -1; if (lastRow > 1) { const dateValues = sheet.getRange(2, dateColIdx + 1, lastRow - 1, 1).getValues(); for (let i = 0; i < dateValues.length; i++) { if (String(dateValues[i][0]).trim() === date) { targetRow = i + 2; break; } } } if (targetRow === -1) { targetRow = sheet.getLastRow() + 1; sheet.getRange(targetRow, dateColIdx + 1).setValue(date); } sheet.getRange(targetRow, fieldColIdx + 1).setNumberFormat('@'); sheet.getRange(targetRow, fieldColIdx + 1).setValue(value); return targetRow; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
